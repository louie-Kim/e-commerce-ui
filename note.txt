??
왼쪽 값이 null 또는 undefined일 때만 오른쪽 값을 실행(반환)하는 연산자

let value = null ?? '기본값';
console.log(value); // "기본값"

value = undefined ?? '기본값';
console.log(value); // "기본값"

value = 0 ?? '기본값';
console.log(value); // 0  // 0은 null이나 undefined가 아니므로 왼쪽 값 사용

findIndex
배열에서 조건을 만족하는 
요소의 "인덱스 번호"를 반환

const arr = [10, 20, 30];

arr.findIndex(num => num === 20); 
// 결과: 1  (20은 배열의 두 번째 요소 → 인덱스 1)

arr.findIndex(num => num > 25);
// 결과: 2  (30이 조건을 처음 만족 → 인덱스 2)

arr.findIndex(num => num === 50);
// 결과: -1 (조건을 만족하는 값 없음)

filter(): 조건이 참 인거만 남긴다.

const fruits = ['apple', 'banana', 'orange'];

// banana만 제외
const noBanana = fruits.filter(fruit => fruit !== 'banana');
console.log(noBanana); // ['apple', 'orange']
